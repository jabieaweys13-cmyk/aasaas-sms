AASAAS SMS — app-kaaga gaarka ah ee SMS-ka lacagta u gudbiya websaydka
=====================================================================

WAXA UU QABTO
- SMS kasta oo yimaada (192, eDahab, JEEB, Somnet…) isla markiiba wuxuu u dirayaa websaydka.
- Haddii internet jirin ama websaydku xiran yahay, wuu kaydinayaa oo mar kale ayuu isku dayayaa ilaa uu gaaro.
- Fariin kasta waxaad ku aragtaa app-ka: ✅ La diray · ⏳ Sugaya · ❌ Khalad.
- "📥 Soo dir lacagihii hore": lacagaha 3-dii maalmood ee la soo dhaafay mar kale ayuu u dirayaa (websaydku kuwa uu hore u haystay ma xisaabiyo mar labaad).
- Ogeysiis joogto ah ayuu leeyahay si Infinix uusan u xirin.
- Fariimaha shakhsiga ah ee hore lama diro — kaliya kuwa lacagta ("heshay").
- SMS-ka caadiga ah sidiisa ayuu u imanayaa app-ka fariimaha; app-kani kaliya wuu akhriyaa oo gudbiyaa.

SIDA LOO RAKIBO (Android Studio + USB)
Sababta USB: Play Protect wuu xannibaa apps-ka SMS ee internetka laga soo dejiyo,
laakiin ma xannibo app aad adigu kombiyuutarkaaga ku dhistay oo USB ku rakibtay.

1. Faylka zip-ka ka saar (Extract) kombiyuutarka.
2. Android Studio → File → Open → dooro galka "AasaasSMS" → OK.
   Sug inta "Gradle sync" dhammaanayo (markii ugu horreysay internet ayuu u baahan yahay).
   Haddii uu ku weydiiyo "Upgrade" ama "Trust project", taabo Trust / OK.
3. Taleefanka: Settings → About phone → taabo "Build number" 7 jeer → Developer options →
   shid "USB debugging". Ku xir USB, taleefankana ka oggolow "Allow USB debugging".
4. Android Studio kor: dooro taleefankaaga → taabo ▶ Run.
   App-ka "Aasaas SMS" wuu ku rakibmayaa oo furmayaa.

MARKA APP-KU FURMO
1. Oggolow: SMS iyo Notifications.
2. Websaydka → Lacagaha EVC → koobi link-ga (sanduuqa MacroDroid, wuxuu leeyahay key=…).
   App-ka → "📋 Dheji" → "💾 Kaydi".
3. Taabo "🔎 Tijaabi link-ga" → waa inuu yiraahdo ✅ Link-gu wuu shaqaynayaa.
4. Taabo "🔋 Battery" → Allow. Sidoo kale: App info → Battery → Unrestricted, Auto-start → ON.
5. Taabo "📥 Soo dir lacagihii hore" si lacagihii seegay ay u gaaraan websaydka.

MacroDroid: waad dami kartaa, ama ha ahaado kaab. Websaydku lacag laba jeer ma xisaabinayo.

import android.content.BroadcastReceiver;

/** After the phone restarts (or the app is updated): start again and send anything that is waiting. */
public class BootReceiver extends BroadcastReceiver {
    @Override
    public void onReceive(Context context, Intent intent) {
        Context app = context.getApplicationContext();
        SendJob.schedulePeriodic(app);
        SendJob.scheduleNow(app);

    }

package art.aasaastv.sms;

import android.content.ContentValues;
import android.content.Context;
import android.database.Cursor;
import android.database.sqlite.SQLiteDatabase;
import android.database.sqlite.SQLiteOpenHelper;

import java.util.ArrayList;
import java.util.List;

/** Local store of every SMS the phone received, with its sending status. */
public class Db extends SQLiteOpenHelper {

    public static final int PENDING = 0;
    public static final int SENT = 1;
    public static final int FAILED = 2;

    public static class Msg {
        public long id;
        public String sender;
        public String body;
        public long ts;
        public int status;
        public String info;
        public int tries;
    }

    private static Db inst;

    public static synchronized Db get(Context c) {
        if (inst == null) {
            inst = new Db(c.getApplicationContext());
        }
        return inst;
    }

    private Db(Context c) {
        super(c, "aasaas_sms.db", null, 1);
    }

    @Override
    public void onCreate(SQLiteDatabase db) {
        db.execSQL("CREATE TABLE msgs (id INTEGER PRIMARY KEY AUTOINCREMENT, sender TEXT, body TEXT, ts INTEGER, "
                + "status INTEGER DEFAULT 0, info TEXT DEFAULT '', tries INTEGER DEFAULT 0, UNIQUE(sender, body, ts))");
    }

    @Override
    public void onUpgrade(SQLiteDatabase db, int oldV, int newV) {
    }

    /** Adds a message. Returns false if the same message (sender, text, time) is already stored. */
    public synchronized boolean add(String sender, String body, long ts) {
        ContentValues v = new ContentValues();
        v.put("sender", sender == null ? "" : sender);
        v.put("body", body == null ? "" : body);
        v.put("ts", ts);
        v.put("status", PENDING);
        return getWritableDatabase().insertWithOnConflict("msgs", null, v, SQLiteDatabase.CONFLICT_IGNORE) != -1;
    }

    public synchronized void mark(long id, int status, String info, boolean countTry) {
        SQLiteDatabase db = getWritableDatabase();
        db.execSQL("UPDATE msgs SET status = ?, info = ?, tries = tries + ? WHERE id = ?",
                new Object[]{status, info == null ? "" : info, countTry ? 1 : 0, id});
    }

    public synchronized void retry(long id) {
        getWritableDatabase().execSQL("UPDATE msgs SET status = 0, tries = 0, info = '' WHERE id = ?", new Object[]{id});
    }

    public synchronized void retryAllFailed() {
        getWritableDatabase().execSQL("UPDATE msgs SET status = 0, tries = 0 WHERE status = 2");
    }

    /** Messages that still need to reach the website (oldest first). */
    public synchronized List<Msg> unsent(int limit) {
        return query("status != 1 AND tries < 60", "id ASC", limit);
    }

    public synchronized List<Msg> recent(int limit) {
        return query(null, "ts DESC, id DESC", limit);
    }

    public synchronized int count(int status) {
        Cursor c = getReadableDatabase().rawQuery("SELECT COUNT(*) FROM msgs WHERE status = ?", new String[]{String.valueOf(status)});
        try {
            return c.moveToFirst() ? c.getInt(0) : 0;
        } finally {
            c.close();
        }
    }

    private List<Msg> query(String where, String order, int limit) {
        List<Msg> out = new ArrayList<>();
        Cursor c = getReadableDatabase().query("msgs", null, where, null, null, null, order, String.valueOf(limit));
        try {
            while (c.moveToNext()) {
                Msg m = new Msg();
                m.id = c.getLong(c.getColumnIndexOrThrow("id"));
                m.sender = c.getString(c.getColumnIndexOrThrow("sender"));
                m.body = c.getString(c.getColumnIndexOrThrow("body"));
                m.ts = c.getLong(c.getColumnIndexOrThrow("ts"));
                m.status = c.getInt(c.getColumnIndexOrThrow("status"));
                m.info = c.getString(c.getColumnIndexOrThrow("info"));
                m.tries = c.getInt(c.getColumnIndexOrThrow("tries"));
                out.add(m);
            }
        } finally {
            c.close();
        }
        return out;
    }
}

package art.aasaastv.sms;

import android.content.Context;
import android.content.SharedPreferences;

import java.io.ByteArrayOutputStream;
import java.io.InputStream;
import java.io.OutputStream;
import java.net.HttpURLConnection;
import java.net.URL;
import java.net.URLEncoder;
import java.nio.charset.StandardCharsets;
import java.util.List;

/** Sends stored SMS to the website: POST text/plain, the same request the plugin expects from MacroDroid. */
public class Forwarder {

    public static final String DEFAULT_URL = "https://aasaastv.art/wp-json/aasaas-pay/v1/sms?key=";

    private static final Object LOCK = new Object();

    public static SharedPreferences prefs(Context c) {
        return c.getSharedPreferences("aasaas", Context.MODE_PRIVATE);
    }

    public static String url(Context c) {
        return prefs(c).getString("url", "").trim();
    }

    /** True when the link has a key after "key=". */
    public static boolean ready(Context c) {
        String u = url(c);
        int k = u.indexOf("key=");
        return u.startsWith("https://") && k > 0 && k + 4 < u.length() && u.charAt(k + 4) != '&';
    }

    /** Sender filter: empty = every SMS. Otherwise comma separated, e.g. "192, eDahab". */
    public static boolean wanted(Context c, String sender) {
        String f = prefs(c).getString("senders", "").trim();
        if (f.isEmpty()) {
            return true;
        }
        String s = sender == null ? "" : sender.toLowerCase();
        for (String part : f.split(",")) {
            String p = part.trim().toLowerCase();
            if (!p.isEmpty() && (s.equals(p) || s.endsWith(p))) {
                return true;
            }
        }
        return false;
    }

    /** Sends every message that has not reached the website yet. Returns how many are still waiting. */
    public static int sendPending(Context c) {
        synchronized (LOCK) {
            Db db = Db.get(c);
            if (!ready(c)) {
                return db.unsent(500).size();
            }
            List<Db.Msg> list = db.unsent(50);
            for (Db.Msg m : list) {
                Result r = post(c, m.sender, m.body);
                if (r.ok) {
                    db.mark(m.id, Db.SENT, r.info, true);
                } else {
                    db.mark(m.id, r.permanent ? Db.FAILED : Db.PENDING, r.info, true);
                    if (r.offline) {
                        break; // no internet: stop and try again later
                    }
                }
            }
            return db.unsent(500).size();
        }
    }

    public static class Result {
        public boolean ok;
        public boolean permanent;
        public boolean offline;
        public String info = "";
    }

    public static Result post(Context c, String sender, String body) {
        Result r = new Result();
        HttpURLConnection con = null;
        try {
            String from = URLEncoder.encode(sender == null ? "" : sender, "UTF-8");
            String u = url(c);
            if (u.contains("{sms_number}")) {
                u = u.replace("{sms_number}", from);
            } else if (!u.contains("from=")) {
                u = u + (u.contains("?") ? "&" : "?") + "from=" + from;
            }
            byte[] data = (body == null ? "" : body).getBytes(StandardCharsets.UTF_8);
            con = (HttpURLConnection) new URL(u).openConnection();
            con.setConnectTimeout(15000);
            con.setReadTimeout(20000);
            con.setRequestMethod("POST");
            con.setDoOutput(true);
            con.setRequestProperty("Content-Type", "text/plain; charset=utf-8");
            con.setRequestProperty("User-Agent", "AasaasSMS/1.0");
            con.setFixedLengthStreamingMode(data.length);
            OutputStream os = con.getOutputStream();
            os.write(data);
            os.close();
            int code = con.getResponseCode();
            String resp = read(code >= 400 ? con.getErrorStream() : con.getInputStream());
            String status = between(resp, "\"status\":\"", "\"");
            if (code == 200 && resp.contains("\"ok\":true")) {
                r.ok = true;
                r.info = status.isEmpty() ? "200" : status;
            } else {
                r.info = code + (status.isEmpty() ? "" : " " + status);
                // 403 = wrong key, 404 = plugin off / wrong link: retrying the same thing will not help.
                r.permanent = code == 403 || code == 404 || code == 400;
            }
        } catch (java.net.UnknownHostException | java.net.ConnectException | java.net.SocketTimeoutException e) {
            r.offline = true;
            r.info = "internet ma jiro";
        } catch (Exception e) {
            r.info = e.getClass().getSimpleName();
        } finally {
            if (con != null) {
                con.disconnect();
            }
        }
        return r;
    }

    private static String read(InputStream in) {
        if (in == null) {
            return "";
        }
        try {
            ByteArrayOutputStream b = new ByteArrayOutputStream();
            byte[] buf = new byte[2048];
            int n;
            while ((n = in.read(buf)) > 0 && b.size() < 8192) {
                b.write(buf, 0, n);
            }
            in.close();
            return b.toString("UTF-8");
        } catch (Exception e) {
            return "";
        }
    }

    private static String between(String s, String a, String b) {
        int i = s.indexOf(a);
        if (i < 0) {
            return "";
        }
        int j = s.indexOf(b, i + a.length());
        return j < 0 ? "" : s.substring(i + a.length(), j);
    }
}

package art.aasaastv.sms;

import android.app.Notification;
import android.app.NotificationChannel;
import android.app.NotificationManager;
import android.app.PendingIntent;
import android.app.Service;
import android.content.Context;
import android.content.Intent;
import android.content.IntentFilter;
import android.provider.Telephony;
import android.content.pm.ServiceInfo;
import android.os.Build;
import android.os.Handler;
import android.os.IBinder;
import android.os.Looper;

/**
 * A small always-on notification. It tells phones like Infinix/Tecno not to kill the app,
 * so SMS keep reaching the website even at night.
 */
public class KeepService extends Service {

    private static final String CH = "keep";
    private final Handler handler = new Handler(Looper.getMainLooper());
    private SmsReceiver live;
    private final Runnable tick = new Runnable() {
        @Override
        public void run() {
            final Context app = getApplicationContext();
            new Thread(() -> Forwarder.sendPending(app)).start();
            handler.postDelayed(this, 5 * 60 * 1000L);
        }
    };

    public static void start(Context c) {
        if (!Forwarder.prefs(c).getBoolean("keep", true)) {
            return;
        }
        try {
            c.startForegroundService(new Intent(c, KeepService.class));
        } catch (Exception ignored) {
            // Android may refuse when started from the background; the SMS receiver still works.
        }
    }

    public static void stop(Context c) {
        c.stopService(new Intent(c, KeepService.class));
    }

    @Override
    public int onStartCommand(Intent intent, int flags, int startId) {
        NotificationManager nm = (NotificationManager) getSystemService(NOTIFICATION_SERVICE);
        if (nm != null && nm.getNotificationChannel(CH) == null) {
            NotificationChannel ch = new NotificationChannel(CH, "Aasaas SMS shaqaynaya", NotificationManager.IMPORTANCE_LOW);
            ch.setShowBadge(false);
            nm.createNotificationChannel(ch);
        }
        PendingIntent open = PendingIntent.getActivity(this, 0, new Intent(this, MainActivity.class),
                PendingIntent.FLAG_IMMUTABLE | PendingIntent.FLAG_UPDATE_CURRENT);
        Notification n = new Notification.Builder(this, CH)
                .setSmallIcon(android.R.drawable.stat_notify_chat)
                .setContentTitle("Aasaas SMS waa shaqaynayaa ✅")
                .setContentText("Fariimaha lacagta waxay si toos ah u aadayaan websaydka")
                .setContentIntent(open)
                .setOngoing(true)
                .build();
        try {
            if (Build.VERSION.SDK_INT >= 34) {
                startForeground(7, n, ServiceInfo.FOREGROUND_SERVICE_TYPE_SPECIAL_USE);
            } else {
                startForeground(7, n);
            }
        } catch (Exception e) {
            stopSelf();
            return START_NOT_STICKY;
        }
        // Listen for SMS from inside this always-running service too: phones like Infinix/Tecno
        // block the manifest receiver of apps without "Auto-start", but not a live one.
        if (live == null) {
            live = new SmsReceiver();
            IntentFilter f = new IntentFilter(Telephony.Sms.Intents.SMS_RECEIVED_ACTION);
            f.setPriority(999);
            try {
                if (Build.VERSION.SDK_INT >= 33) {
                    registerReceiver(live, f, "android.permission.BROADCAST_SMS", null, Context.RECEIVER_EXPORTED);
                } else {
                    registerReceiver(live, f, "android.permission.BROADCAST_SMS", null);
                }
            } catch (Exception e) {
                live = null;
            }
        }
        handler.removeCallbacks(tick);
        handler.postDelayed(tick, 60 * 1000L);
        return START_STICKY;
    }

    @Override
    public void onDestroy() {
        handler.removeCallbacks(tick);
        if (live != null) {
            try {
                unregisterReceiver(live);
            } catch (Exception ignored) {
            }
            live = null;
        }
        super.onDestroy();
    }

    @Override
    public IBinder onBind(Intent intent) {
        return null;
    }
}

    /** Re-sends payment SMS already in the phone's inbox (the website ignores ones it already has). */
    private void importOld() {
        if (!granted(Manifest.permission.READ_SMS)) {
            askPermissions();
            return;
        }
        new Thread(() -> {
            int n = 0;
            long since = System.currentTimeMillis() - 3L * 24 * 60 * 60 * 1000;
            Cursor c = null;
            try {
                c = getContentResolver().query(Telephony.Sms.Inbox.CONTENT_URI,
                        new String[]{Telephony.Sms.ADDRESS, Telephony.Sms.BODY, Telephony.Sms.DATE},
                        Telephony.Sms.DATE + " > ?", new String[]{String.valueOf(since)}, Telephony.Sms.DATE + " ASC");
                while (c != null && c.moveToNext()) {
                    String from = c.getString(0);
                    String body = c.getString(1);
                    long ts = c.getLong(2);
                    if (body == null || !looksLikePayment(body) || !Forwarder.wanted(this, from)) {
                        continue;
                    }
                    if (Db.get(this).add(from == null ? "" : from, body, ts)) {
                        n++;
                    }
                }
            } catch (Exception ignored) {
            } finally {
                if (c != null) {
                    c.close();
                }
            }
            final int added = n;
            runOnUiThread(() -> toast(added == 0 ? "Wax cusub lama helin" : "📥 " + added + " fariin ayaa la dirayaa"));
            if (added > 0) {
                SendJob.scheduleNow(this);
                Forwarder.sendPending(this);
            }
        }).start();
    }

    }

    private void askPermissions() {
        List<String> need = new ArrayList<>();
        need.add(Manifest.permission.RECEIVE_SMS);
        need.add(Manifest.permission.READ_SMS);
        if (Build.VERSION.SDK_INT >= 33) {
            need.add(Manifest.permission.POST_NOTIFICATIONS);
        }
        List<String> missing = new ArrayList<>();
        for (String p : need) {
            if (!granted(p)) {
                missing.add(p);
            }
        }
        if (missing.isEmpty()) {
            return;
        }
        if (!shouldShowRequestPermissionRationale(Manifest.permission.RECEIVE_SMS) && asked()) {
            // Refused before: open the app settings so the person can allow SMS by hand.
            startActivity(new Intent(Settings.ACTION_APPLICATION_DETAILS_SETTINGS, Uri.parse("package:" + getPackageName())));
            return;
        }
        Forwarder.prefs(this).edit().putBoolean("asked", true).apply();
        requestPermissions(missing.toArray(new String[0]), 1);
    }

    private boolean asked() {
        return Forwarder.prefs(this).getBoolean("asked", false) && !granted(Manifest.permission.RECEIVE_SMS);
    }

package art.aasaastv.sms;

import android.app.job.JobInfo;
import android.app.job.JobParameters;
import android.app.job.JobScheduler;
import android.app.job.JobService;
import android.content.ComponentName;
import android.content.Context;

/** Retries messages that could not be sent (no internet, site down) as soon as the phone is online again. */
public class SendJob extends JobService {

    private static final int NOW_ID = 1;
    private static final int PERIODIC_ID = 2;

    public static void scheduleNow(Context c) {
        JobScheduler js = (JobScheduler) c.getSystemService(Context.JOB_SCHEDULER_SERVICE);
        if (js == null) {
            return;
        }
        js.schedule(new JobInfo.Builder(NOW_ID, new ComponentName(c, SendJob.class))
                .setRequiredNetworkType(JobInfo.NETWORK_TYPE_ANY)
                .setBackoffCriteria(30_000, JobInfo.BACKOFF_POLICY_EXPONENTIAL)
                .setPersisted(true)
                .build());
    }

    /** A check every 15 minutes, in case something was missed. */
    public static void schedulePeriodic(Context c) {
        JobScheduler js = (JobScheduler) c.getSystemService(Context.JOB_SCHEDULER_SERVICE);
        if (js == null) {
            return;
        }
        js.schedule(new JobInfo.Builder(PERIODIC_ID, new ComponentName(c, SendJob.class))
                .setRequiredNetworkType(JobInfo.NETWORK_TYPE_ANY)
                .setPeriodic(15 * 60 * 1000L)
                .setPersisted(true)
                .build());
    }

    @Override
    public boolean onStartJob(final JobParameters params) {
        final Context app = getApplicationContext();
        new Thread(() -> {
            int left = Forwarder.sendPending(app);
            // Ask Android to run again (with back-off) if some messages are still waiting.
            jobFinished(params, left > 0 && params.getJobId() == NOW_ID && Forwarder.ready(app));
        }).start();
        return true;
    }

    @Override
    public boolean onStopJob(JobParameters params) {
        return true;
    }
}

package art.aasaastv.sms;

import android.content.BroadcastReceiver;
import android.content.Context;
import android.content.Intent;
import android.provider.Telephony;
import android.telephony.SmsMessage;

import java.util.LinkedHashMap;
import java.util.Map;

/** Runs for every incoming SMS (even when the app is closed): store it, then send it to the website. */
public class SmsReceiver extends BroadcastReceiver {

    @Override
    public void onReceive(Context context, Intent intent) {
        if (!Telephony.Sms.Intents.SMS_RECEIVED_ACTION.equals(intent.getAction())) {
            return;
        }
        SmsMessage[] parts = Telephony.Sms.Intents.getMessagesFromIntent(intent);
        if (parts == null || parts.length == 0) {
            return;
        }
        // Long SMS arrive in several parts: join them per sender.
        Map<String, StringBuilder> bodies = new LinkedHashMap<>();
        Map<String, Long> times = new LinkedHashMap<>();
        for (SmsMessage p : parts) {
            if (p == null) {
                continue;
            }
            String from = p.getDisplayOriginatingAddress();
            if (from == null) {
                from = "";
            }
            StringBuilder sb = bodies.get(from);
            if (sb == null) {
                sb = new StringBuilder();
                bodies.put(from, sb);
                times.put(from, p.getTimestampMillis() > 0 ? p.getTimestampMillis() : System.currentTimeMillis());
            }
            String b = p.getDisplayMessageBody();
            if (b != null) {
                sb.append(b);
            }
        }
        final Context app = context.getApplicationContext();
        Forwarder.prefs(app).edit().putLong("last_rx", System.currentTimeMillis()).apply();
        boolean added = false;
        for (Map.Entry<String, StringBuilder> e : bodies.entrySet()) {
            if (Forwarder.wanted(app, e.getKey())) {
                added |= Db.get(app).add(e.getKey(), e.getValue().toString(), times.get(e.getKey()));
            }
        }
        if (!added) {
            return;
        }
        SendJob.scheduleNow(app);
        final PendingResult pr = goAsync();
        new Thread(() -> {
            try {
                Forwarder.sendPending(app);
            } finally {
                pr.finish();
            }
        }).start();
    }
}
